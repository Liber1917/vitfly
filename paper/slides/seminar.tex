% ─── SLIDE 4: MAMBA ───
\begin{frame}{方案：状态空间模型 (Mamba)}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{itemize}
\item \textbf{选择性SSM}：输入驱动的动态状态转移
\item 复杂度：\pos{O(n)} vs 注意力 O(n$^2$)
\item 硬件感知并行扫描 — GPU高效
\item 选择性 = 动态信息流 = 保留关键，遗忘冗余
\end{itemize}

\begin{block}{关键机制}
\[
 h_t = \mathbf{\bar{A}}h_{t-1} + \mathbf{\bar{B}}x_t,\quad
 y_t = \mathbf{C}h_t
 \]
 \[
 \mathbf{B},\mathbf{C},\Delta = f_\text{selective}(x_t)
 \]
\end{block}

\column{0.45\textwidth}
\begin{block}{三个核心研究问题}
\begin{enumerate}
\item \textbf{架构适用性}\\
哪种Mamba设计避障最优？
\item \textbf{知识迁移可行性}\\
ViT教师 $\to$ Mamba学生有效吗？
\item \textbf{蒸馏边界条件}\\
什么时候蒸馏成功/失败？
\end{enumerate}
\end{block}
\end{columns}
\end{frame}

% ─── SLIDE 7: CORE DISCOVERY ───
\begin{frame}{核心发现：空间结构 = 蒸馏边界条件}
\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{E-SSM 对照实验}
\begin{itemize}
\item \pos{CNN编码器(E)}：\\
$\mathcal{L}_\text{feat}=0.85$，\textbf{蒸馏成功}
\item \con{SSM编码器(E-SSM)}：\\
$\mathcal{L}_\text{feat}=10.58$\\
$\mathcal{L}_\text{GT}$: 0.02 $\to$ 0.83，\textbf{完全失败}
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{alertblock}{根因分析}
\centering
\begin{tikzpicture}[node distance=0.3cm,font=\tiny]
\node[draw,rounded corners,fill=positive!15,text width=2.5cm,align=center] (C) {CNN\\保留2D空间结构};
\node[draw,rounded corners,fill=negative!15,text width=2.5cm,align=center,below=of C] (S) {SSM\\扫描展平为1D序列};
\node[right=0.1cm of C,text width=2cm,align=left] (R1) {$\to$ MSE对齐有效};
\node[right=0.1cm of S,text width=2cm,align=left] (R2) {$\to$ 逐点对应被破坏};
\draw[->] (C)--(R1); \draw[->] (S)--(R2);
\end{tikzpicture}
\end{alertblock}
\end{columns}
\vspace{0.2cm}
\begin{block}{\HL{设计原则}}
\textbf{跨架构蒸馏的必要前提：学生编码器必须保留空间结构。}
\end{block}
\end{frame}

% ─── SLIDE 10: ABLATION ───
\begin{frame}{消融洞察：设计因素排序}
\begin{table}
\centering\footnotesize
\begin{tabular}{lccc}
\toprule
设计因素 & 最佳配置 & 碰撞范围 & 影响力 \\
\midrule
\HL{编码器类型} & CNN/混合 & 1--5 & \HL{★★★★★} \\
知识蒸馏 & 三分量 & 1--DNF & ★★★★☆ \\
数据增强 & CNN架构感知 & 1--5 & ★★★☆☆ \\
损失权重 & 均衡 & 1--4 & ★★☆☆☆ \\
多步预测 & T=1 & 1--5 & ★★☆☆☆ \\
\bottomrule
\end{tabular}
\end{table}
\begin{itemize}
\item 编码器 $>$ 蒸馏 $>$ 增强 $>$ 权重 $>$ 时序长度
\item T=1单步始终最优 — 当前帧精确感知 $>$ 长程记忆
\item 数据增强：\pos{CNN有益(B+:3$\to$1)}，\con{SSM有害(D:2$\to$5)}
\item G\_basic (CNN+MLP) 仅比E多1碰撞，但快9.6倍 — SSM时序头边际贡献
\end{itemize}
\end{frame}

% ─── SLIDE 14: SUMMARY ───
\begin{frame}{总结}
\begin{enumerate}
\item \HL{首次系统评估6种Mamba架构在四旋翼控制中的适用性}\\
\footnotesize SS2D·MambaVision·CNN三种编码器 + 四种时序头全组合对比

\item \HL{揭示跨架构蒸馏的核心边界条件}\\
\footnotesize 编码器保留空间结构 $\to$ 有效；SSM扫描展平 $\to$ 完全失败

\item \HL{设计优先级：编码器 $>$ 蒸馏 $>$ 数据增强 $>$ 损失权重 $>$ 多步预测}
\end{enumerate}
