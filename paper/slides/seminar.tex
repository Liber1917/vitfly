        {$L_\text{distill}$\\\footnotesize{输出蒸馏}};
      \node[draw,rounded corners,fill=blue!35,below=of T,xshift=2cm,minimum width=2cm] (L3)
        {$L_\text{GT}$\\\footnotesize{真值监督}};

      % Student
      \node[draw,rounded corners,fill=green!20,below=of L2,xshift=0cm,minimum width=7cm,minimum height=1.2cm] (S)
        {\textbf{Mamba学生} \quad A: VMamba+LSTM \quad B/B+: MambaVision \quad C: CNN+Mamba3 \quad D: STH-Mamba \quad E: DecisionMamba};

      \draw[->,thick] (T) -- (L1); \draw[->,thick] (T) -- (L2); \draw[->,thick] (T) -- (L3);
      \draw[->,thick] (L1) -- (S); \draw[->,thick] (L2) -- (S); \draw[->,thick] (L3) -- (S);
    \end{tikzpicture}
  \end{center}

  \begin{block}{损失函数}
    \[
    \mathcal{L} = \alpha\underbrace{\|f_\text{student} - f_\text{teacher}\|^2}_{\text{特征对齐}}
                  + \beta\underbrace{\|\hat{y}_\text{student} - \hat{y}_\text{teacher}\|^2}_{\text{输出蒸馏}}
                  + \gamma\underbrace{\|\hat{y}_\text{student} - y_\text{GT}\|^2}_{\text{真值监督}}
    \]
    ($\alpha=\beta=\gamma=1$ 时最优)
  \end{block}
\end{frame}

% ═══════ SLIDE 8: 核心发现 ═══════
\begin{frame}{核心发现：编码器必须保留空间结构}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{E-SSM对照实验}
      \begin{itemize}
        \item E (CNN编码器)：$\mathcal{L}_\text{feat}=0.85$，蒸馏成功
        \item E-SSM (替换为SSM编码器，相同输出维度)：\\
              $\mathcal{L}_\text{feat}=10.58$\\
              $\mathcal{L}_\text{GT}$：0.02 $\to$ 0.83\\
              \textbf{蒸馏完全失败}
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{alertblock}{为什么失败}
      \begin{center}
        \begin{tikzpicture}
          \node[draw,rounded corners,fill=green!15,minimum width=2.2cm] (CNN) {CNN编码器\\[2pt]\footnotesize{保留2D空间结构}};
          \node[draw,rounded corners,fill=red!15,minimum width=2.2cm,right=of CNN] (SSM) {SSM编码器\\[2pt]\footnotesize{扫描展平为1D序列}};
          \node[draw,rounded corners,fill=gray!10,below=of CNN,minimum width=2.2cm] (R1) {MSE对齐有效\\[2pt]\footnotesize{特征空间兼容}};
          \node[draw,rounded corners,fill=gray!10,below=of SSM,minimum width=2.2cm] (R2) {MSE对齐失败\\[2pt]\footnotesize{逐点对应被破坏}};
          \draw[->] (CNN) -- (R1); \draw[->] (SSM) -- (R2);
        \end{tikzpicture}
      \end{center}
    \end{alertblock}
  \end{columns}

  \begin{block}{一般性设计原则}
    \textbf{跨架构蒸馏的必要前提：学生编码器必须保留空间结构。}
    这一原则不仅适用于本研究的ViT$\to$Mamba场景，
    对未来所有跨架构蒸馏的编码器选型都具有指导意义。
  \end{block}
\end{frame}

  \end{alertblock}
\end{frame}

% ═══════ SLIDE 11: 消融洞察 ═══════
\begin{frame}{消融实验：什么因素最重要}
  \begin{table}
    \centering
    \begin{tabular}{lcccl}
      \toprule
      \textbf{设计因素} & \textbf{最佳配置} & \textbf{碰撞范围} & \textbf{影响} \\
      \midrule
      编码器类型 & CNN/混合（保留空间结构） & 1--5 & ★★★★★ \\
      知识蒸馏 & 三分量蒸馏 ($\alpha=\beta=\gamma=1$) & 1--DNF & ★★★★☆ \\
      数据增强 & 仅对CNN编码器启用 & 1--5 & ★★★☆☆ \\
      损失权重 & 默认均衡权重 & 1--4 & ★★☆☆☆ \\
      多步预测 & T=1（单步） & 1--5 & ★★☆☆☆ \\
      \bottomrule
    \end{tabular}
  \end{table}

  \begin{block}{关键发现}
    \begin{itemize}
      \item \textbf{编码器类型} 的优先权超过所有其他因素之和
      \item \textbf{单步预测(T=1)} 在避障中始终优于多步——当前帧的精确感知 $>$ 长程时序记忆
      \item \textbf{数据增强} 的效果高度依赖架构类型：\\
            CNN编码器有益（B+：3$\to$1），SSM编码器有害（D：2$\to$5）
    \end{itemize}
  \end{block}
\end{frame}
